🚀 사용법 가이드

1. 간단한 버전 확인

console.log($.http.version); // "1.0.0"

2. 상세 정보 객체 가져오기

const info = $.http.getVersion(); console.log(info); /* { version: "1.0.0", buildDate: "2025-08-01", author: "jaehoosu", features: [...] } */

3. 예쁜 형태로 콘솔 출력

$.http.printVersion(); // 콘솔에 박스 형태로 예쁘게 출력됩니다

4. 런타임에서 호환성 확인

// 최소 버전 확인 function checkMinVersion(minVersion) { const current = $.http.version; const [curMajor, curMinor, curPatch] = current.split('.').map(Number); const [minMajor, minMinor, minPatch] = minVersion.split('.').map(Number); if (curMajor > minMajor) return true; if (curMajor === minMajor && curMinor > minMinor) return true; if (curMajor === minMajor && curMinor === minMinor && curPatch >= minPatch) return true; return false; } if (checkMinVersion('1.0.0')) { console.log('호환 가능한 버전입니다!'); } else { console.warn('버전을 업데이트해주세요.'); }
